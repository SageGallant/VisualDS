Insertion

function insertAtBeginning(value) {
    const newNode = new Node(value);
    newNode.next = head;
    head = newNode;
}

Deletion

function deleteFromEnd() {
    if (!head) return null;
    if (!head.next) {
        head = null;
        return;
    }
    let current = head;
    while (current.next.next) {
        current = current.next;
    }
    current.next = null;
}

Implementation Details

Insertion

Beginning: O(1)
End: O(n)
At Position: O(n)

Deletion

Beginning: O(1)
End: O(n)
At Position: O(n)
